Look at the actual syntax

Selected lines from the authored fixture. The full fixture connects the basis, metric, shared markers and points. A caller selects a reader and binds its request; these declarations describe the inputs.

let $contour = crystal(orthogonal($origin));
let $centre_representation = $origin -> $centre;
let $rims = frame($m0 -> ($centre -> $p0), $m1 -> ($centre -> $p1),
    $m2 -> ($centre -> $p2), $m3 -> ($centre -> $p3));
let $cycle = frame($m0 -> $m1, $m1 -> $m2, $m2 -> $m3, $m3 -> $m0);
let $surface = group($centre_representation, $contour, $metric, $rims, $cycle);
let $walk = group($surface, $m0, span($steps, 0, 1));
root group(?traversal, $points);
